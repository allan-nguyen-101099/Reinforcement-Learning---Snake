// -------------------------------------------
//              Iterators
// -------------------------------------------

// An iterator is anything that implements the Iterator trait.
// Iterators are LAZY — they do nothing until consumed.
// They produce one item at a time via the next() method.

// The Iterator trait (simplified from std):
// trait Iterator {
//     type Item;                          // associated type — the type of items produced
//     fn next(&mut self) -> Option<Self::Item>;  // returns Some(item) or None when done
// }

// -------------------------------------------
// 1. Creating Iterators from Collections
// -------------------------------------------

fn creating_iterators() {
    let v = vec![1, 2, 3];

    // .iter()      — yields &T      (immutable references) — collection stays intact
    // .iter_mut()  — yields &mut T  (mutable references)   — collection stays intact
    // .into_iter() — yields T       (owned values)          — collection is consumed

    for x in v.iter() {
        println!("{}", x);    // x is &i32
    }
    println!("{:?}", v);      // v is still usable

    for x in v.into_iter() {
        println!("{}", x);    // x is i32 (owned)
    }
    // println!("{:?}", v);   // ERROR — v was consumed
}

// -------------------------------------------
// 2. Using next() Manually
// -------------------------------------------

fn manual_next() {
    let v = vec![10, 20, 30];
    let mut iter = v.iter();

    println!("{:?}", iter.next());   // Some(10)
    println!("{:?}", iter.next());   // Some(20)
    println!("{:?}", iter.next());   // Some(30)
    println!("{:?}", iter.next());   // None
}

// -------------------------------------------
// 3. Iterator Adaptors — Transform the Iterator (still lazy)
// -------------------------------------------

fn adaptors() {
    let v = vec![1, 2, 3, 4, 5];

    // map — transform each element
    let doubled = v.iter().map(|x| x * 2);          // still lazy!
    let doubled: Vec<i32> = v.iter().map(|x| x * 2).collect();  // now consumed
    println!("{:?}", doubled);    // [2, 4, 6, 8, 10]

    // filter — keep elements matching a condition
    let evens: Vec<&i32> = v.iter().filter(|x| *x % 2 == 0).collect();
    println!("{:?}", evens);      // [2, 4]

    // filter_map — filter and transform in one step
    let results: Vec<i32> = vec!["1", "two", "3", "four"]
        .iter()
        .filter_map(|s| s.parse::<i32>().ok())   // parse, discard errors
        .collect();
    println!("{:?}", results);   // [1, 3]

    // enumerate — pairs each element with its index
    for (i, val) in v.iter().enumerate() {
        println!("index {}: {}", i, val);
    }

    // zip — combine two iterators into pairs
    let names = vec!["Alice", "Bob"];
    let scores = vec![95, 87];
    let paired: Vec<_> = names.iter().zip(scores.iter()).collect();
    println!("{:?}", paired);    // [("Alice", 95), ("Bob", 87)]

    // flat_map — map then flatten one level
    let words = vec!["hello world", "foo bar"];
    let letters: Vec<&str> = words.iter().flat_map(|s| s.split(' ')).collect();
    println!("{:?}", letters);   // ["hello", "world", "foo", "bar"]

    // take — stop after N elements
    let first_three: Vec<&i32> = v.iter().take(3).collect();
    println!("{:?}", first_three);  // [1, 2, 3]

    // skip — skip the first N elements
    let after_two: Vec<&i32> = v.iter().skip(2).collect();
    println!("{:?}", after_two);    // [3, 4, 5]

    // chain — concatenate two iterators
    let a = vec![1, 2];
    let b = vec![3, 4];
    let combined: Vec<&i32> = a.iter().chain(b.iter()).collect();
    println!("{:?}", combined);  // [1, 2, 3, 4]
}

// -------------------------------------------
// 4. Iterator Consumers — Finalize the Iterator
// -------------------------------------------

fn consumers() {
    let v = vec![1, 2, 3, 4, 5];

    // collect — gather into a collection
    let doubled: Vec<i32> = v.iter().map(|x| x * 2).collect();

    // sum / product
    let total: i32 = v.iter().sum();        // 15
    let product: i32 = v.iter().product();  // 120

    // count
    let n = v.iter().count();               // 5

    // fold — accumulate with initial value and closure
    let sum = v.iter().fold(0, |acc, x| acc + x);    // 15

    // find — first match (returns Option<&T>)
    let first_even = v.iter().find(|&&x| x % 2 == 0);  // Some(2)

    // position — index of first match (returns Option<usize>)
    let pos = v.iter().position(|&x| x == 3);           // Some(2)

    // any / all — check condition across elements
    let has_even = v.iter().any(|&x| x % 2 == 0);    // true
    let all_pos  = v.iter().all(|&x| x > 0);          // true

    // max / min
    let biggest = v.iter().max();   // Some(5)
    let smallest = v.iter().min();  // Some(1)

    // for_each — like for loop but as a method
    v.iter().for_each(|x| print!("{} ", x));   // 1 2 3 4 5
}

// -------------------------------------------
// 5. Implementing Iterator on a Custom Type
// -------------------------------------------

struct Counter {
    count: u32,
    max: u32,
}

impl Counter {
    fn new(max: u32) -> Self {
        Counter { count: 0, max }
    }
}

impl Iterator for Counter {
    type Item = u32;

    fn next(&mut self) -> Option<Self::Item> {
        if self.count < self.max {
            self.count += 1;
            Some(self.count)
        } else {
            None   // signals exhaustion
        }
    }
}

fn demo_custom() {
    let counter = Counter::new(5);
    let v: Vec<u32> = counter.collect();
    println!("{:?}", v);   // [1, 2, 3, 4, 5]

    // All iterator adaptors work automatically once next() is implemented!
    let sum: u32 = Counter::new(5).sum();
    println!("{}", sum);   // 15
}

/*
---------------------------------------------------------------------------------------------------------------
Concept / Topic              | Explanation
---------------------------------------------------------------------------------------------------------------
Iterator Trait               | - Any type implementing next() is an iterator.
                             | - next() returns Option<Item> — Some(value) or None when done.
                             | - Implementing next() gives you ALL adaptor/consumer methods for free.

.iter()                      | - Borrows collection. Yields &T. Collection usable after.
.iter_mut()                  | - Borrows collection mutably. Yields &mut T.
.into_iter()                 | - Consumes collection. Yields T (owned).

Lazy Evaluation              | - Adaptors (map, filter, etc.) are lazy — do nothing until consumed.
                             | - Consumers (collect, sum, for_each) trigger execution.
                             | - Chaining multiple adaptors incurs only ONE pass — efficient.

Key Adaptors                 | map, filter, filter_map, enumerate, zip, flat_map, take, skip, chain

Key Consumers                | collect, sum, product, count, fold, find, position, any, all, max, min

collect()                    | - Gathers iterator results into a collection (Vec, HashMap, String, etc.)
                             | - Type must be annotated so Rust knows what to collect into.

fold()                       | - Most general consumer — apply a function repeatedly with an accumulator.
                             | - sum, product, count are all special cases of fold.

Custom Iterator              | - impl Iterator for YourType — only need to define next().
                             | - All other methods come from the default implementations in the trait.

C++ Comparison               | - Similar to C++ range-based algorithms and views (C++20 ranges).
                             | - Rust iterators are lazy like C++20 std::views.
---------------------------------------------------------------------------------------------------------------
*/
