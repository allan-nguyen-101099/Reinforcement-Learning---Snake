// -------------------------------------------------
// 			- Associativity
//          - Explicit Boolean in Conditionals
//          - Operator Overloading
// -------------------------------------------------

fn main() {
    // ── Associativity ────────────────────────────────────────────────────────
    // Associativity determines the ORDER in which operators of the SAME precedence
    // are evaluated when they appear in sequence without parentheses.
    // Most operators in Rust are LEFT-ASSOCIATIVE (evaluated left → right).
    //
    // `8 / 4 / 2`   → left-to-right: (8 / 4) / 2 = 2 / 2 = 1
    //                 NOT right-to-left: 8 / (4 / 2) = 8 / 2 = 4
    let x = 8 / 4 / 2; // x = 1  (left-to-right: (8/4)/2)

    let mut y = 42;

    // Chain assignment `x = y = 0` does NOT work in Rust.
    // `y = 0` is a STATEMENT that returns `()` (unit type), not an integer value.
    // So `x = (y = 0)` would try to assign `()` to `x` — a type mismatch error.
    // x = y = 0; // x = (y = 0) // x = ()  ← COMPILE ERROR

    // RIGHT-TO-LEFT associative operators (assignment and compound assignments):
    // =, +=, -=, *=, /=, %=, &=, |=, ^=, <<=, >>=
    // Example: `b += c += 42` first evaluates `c += 42`, then `b += (new c)`

    let mut a = 1;
    let mut b = 2;
    let mut c = 3;
    // b += c += 42; // right-to-left: c = 3+42 = 45, then b = 2+45 = 47

    // ── Explicit Boolean in Conditionals ─────────────────────────────────────
    // Rust requires the condition in `if`, `while`, etc. to be EXPLICITLY boolean.
    // You CANNOT use an integer directly as a condition (unlike C or Python).
    // The expression MUST evaluate to `true` or `false`.
    //
    // `x != 0`      → an explicit boolean comparison — the correct Rust way
    //                 Writing just `if x {}` when x is an integer is a COMPILE ERROR
    let x = 0;
    if x != 0 {}

    // ── Operator Overloading ──────────────────────────────────────────────────
    // Operator overloading means the SAME operator symbol can BEHAVE DIFFERENTLY
    // depending on the TYPES it is applied to.
    // In Rust this is implemented through TRAITS (a form of interface/type contract).
    //
    // `10 + 20`     → `+` on integers: NUMERIC ADDITION → result: 30 (i32)
    let a = 10 + 20; // a = 30

    // `String::from("1") + " 2"` → `+` on String: STRING CONCATENATION
    //   `String::from("1")` → creates an OWNED String containing "1"
    //   `+ " 2"`            → appends the `&str` slice " 2" to the end of the String
    //                         NOTE: left side must be an owned `String`,
    //                               right side must be a `&str` (not another `String`)
    //   Result: a new String "1 2"
    let b = String::from("1") + " 2"; // b = "1 2"
    println!("{}", b); // Output: 1 2
}




/* 
Topic                  | Key Points
-----------------------|-----------------------------------------------------------
Associativity          | Most operators are left-associative (evaluated left → right)
                       | Assignment (=) and bitwise operators are right-associative
                       | Rust disallows chaining (e.g., x = y = 0) due to type safety
                       | Parentheses recommended for clarity
Explicit Boolean       | Rust requires conditions to explicitly return bool
                       | Using integers or strings directly in `if` gives compile-time error
Operator Overloading   | Operators like (+, -, *) may behave differently depending on type
                       | Example: + adds numbers but concatenates Strings
                       | Behavior is implemented using traits
*/
