// -------------------------------------------------
// 			- Option Combinator Chains
// -------------------------------------------------

fn fetch_user(name: &str) -> Option<String> {
    println!("Looking for user: {name}");
    Some(format!("User({})", name))  // format! builds a String with interpolated values
}

fn main() {
    // Pattern 1: map -> filter -> and_then
    // Use case: trim input -> validate -> fetch from DB
    let user = Some("    alice    ")
        .map(str::trim)                       // str::trim as a function pointer — trims whitespace
        .filter(|s| !s.is_empty())            // discard if blank after trimming
        .and_then(|name| fetch_user(name));   // only fetches if still Some

    // Pattern 2: and_then -> and_then -> map
    // Use case: parse input -> look up config -> transform
    let result = Some("42")
        .and_then(|s| s.parse::<u32>().ok())  // parse &str to u32, .ok() converts Result to Option
        .and_then(|s| match s {               // look up config by parsed value
            42 => Some("answer_config".to_string()),
            8080 => Some("web_server_config".to_string()),
            _ => None,  // unknown value — short-circuit the chain
        })
        .map(|config| config.to_uppercase());  // transform the result String

    // Pattern 3: or_else -> or_else -> unwrap_or
    // Use case: try multiple fallback sources, extract a concrete value at the end
    let env_config: Option<String> = None;
    let user_config: Option<String> = Some("/home/user/config.conf".to_string());
    let default_config: Option<String> = Some("/default.config".to_string());

    let path = env_config
        .or_else(|| user_config)      // if env_config is None, try user_config
        .or_else(|| default_config)   // if still None, try default_config
        .unwrap_or("/etc/default.conf".to_string());  // final fallback if ALL are None

    // Pattern 4: filter -> map -> unwrap_or_else
    // Use case: validate a string, parse it, fall back to a default port
    let port = Some("8080")
        .filter(|s| s.chars().all(char::is_numeric))  // reject if not all digits
        .map(|s| s.parse::<u16>().unwrap_or(0))        // parse to u16, 0 on parse failure
        .unwrap_or_else(|| 3000);  // if None (failed filter): default port 3000
}

/*
---------------------------------------------------------------------
Benefits of Combinators
---------------------------------------------------------------------
1. Keeps logic clear and declarative.
2. Eliminates deeply nested match or if let constructs.
3. Ensures early exits on None without additional checks.
4. Encourages lazy evaluation with closures.
---------------------------------------------------------------------
*/
