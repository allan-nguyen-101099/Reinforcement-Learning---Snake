// -------------------------------------------------
// 		Typical Transformations on Vec<Option>
// -------------------------------------------------

fn main() {
    // 1. Unwrap all Some values (panics if any None exists)
    let data = vec![Some(1), Some(2), Some(3)];
    let unwrapped = data
        .into_iter()           // consume Vec, yield Option<i32> values
        .map(|x| x.unwrap() * 2)  // .unwrap() panics if None — only safe when ALL are Some
        .collect::<Vec<i32>>();    // collect into Vec<i32>
    println!("{:?}", unwrapped); // [2, 4, 6]

    // 2. Filter out all None values and unwrap the Some values
    let data = vec![Some(1), None, Some(3), None, Some(5)];
    let filtered = data
        .into_iter()
        // .flatten()  — alternative: flatten() removes None and unwraps Some automatically
        .filter(|x| x.is_some())  // keep only Some variants
        .map(|x| x.unwrap())       // safe to unwrap since filter guarantees Some
        .collect::<Vec<i32>>();
    println!("Filtered data: {:?}", filtered); // [1, 3, 5]

    // 3. Remove None but keep Some still wrapped in Some
    let mut data = vec![Some(1), None, Some(2), None, Some(3)];
    let data = data
        .into_iter()                      // note: .iter() would yield &Option not Option
        .filter(|x| x.is_some())          // keep only Some values
        .collect::<Vec<Option<i32>>>();    // collect keeping them as Option
    println!("Data with None removed {:?}", data);  // [Some(1), Some(2), Some(3)]
}

/*
----------------------------------------------------------------------
Transformation                   | When Used
----------------------------------------------------------------------
Transformation Unwrap all Some   | When used All values are mandatory

Unwrap Some and filter out None  | There are incomplete data

Remove None and keep Some        | Used for perserving Some
----------------------------------------------------------------------
*/
