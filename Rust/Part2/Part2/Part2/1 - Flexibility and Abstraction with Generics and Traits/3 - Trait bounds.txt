// -------------------------------------------
// 			Trait Bounds
// -------------------------------------------

struct Square {
    side: f32,
    line_width: u8,
    color: String,
}

struct Rectangle {
    length: f32,
    width: f32,
    line_width: u8,
    color: String,
}

trait Shape {
    fn area(&self) -> f32;
    fn perimeter(&self) -> f32 {
        println!("Perimeter not implemented, returning dummy value");
        0.0
    }
}

impl Shape for Rectangle {
    fn area(&self) -> f32 {
        let area_of_rect = self.length * self.width;
        println!("Rectangle area: {}", area_of_rect);
        area_of_rect
    }
    fn perimeter(&self) -> f32 {
        let perimeter_of_rect = 2.0 * (self.length + self.width);
        println!("Rectangle Perimeter: {}", perimeter_of_rect);
        perimeter_of_rect
    }
}

impl Shape for Square {
    fn area(&self) -> f32 {
        let area_of_square = self.side * self.side;
        println!("Square area: {}", area_of_square);
        area_of_square
    }
}

fn shape_properties<T>(object: T)  // generic function — T is a type parameter
where
    T: Shape,  // where clause — T must implement Shape (trait bound)
               // without this bound, calling object.area() would fail — compiler doesn't know T has area()
{
    object.area();       // valid because T: Shape guarantees area() exists
    object.perimeter();  // valid because T: Shape guarantees perimeter() exists
}

// `impl Shape` in return position — returns SOME concrete type that implements Shape
// The exact type is hidden from the caller — they only know it has Shape methods
fn returns_shape() -> impl Shape {
    let sq = Square {
        side: 5.0,
        line_width: 5,
        color: String::from("Red"),
    };
    sq  // returning sq — the compiler knows the concrete type is Square, but caller just sees `impl Shape`
    // IMPORTANT: impl Shape in return requires ONE consistent concrete type across all return paths
    // The commented-out code below would FAIL because it tries to return either sq or rect:
    // let rect = Rectangle { ... };
    // let x = false;
    // if x { sq } else { rect }  // ERROR — two different concrete types returned
    // Use Box<dyn Shape> (trait object) instead for conditional returns of different types
}

struct Circle {
    radius: f32,
}
fn main() {
    let r1 = Rectangle {
        width: 5.0,
        length: 4.0,
        line_width: 1,
        color: String::from("Red"),
    };

    let s1 = Square {
        side: 3.2,
        line_width: 1,
        color: String::from("Red"),
    };

    let c1 = Circle { radius: 5.0 };
    shape_properties(r1);
    shape_properties(s1);
    // shape_properties(c1); // Trait bound not satisfied
}

/*
---------------------------------------------------------------------------------------------
Concept/Topic         | Explanation
----------------------|----------------------------------------------------------------------
Trait Bounds          | It restrict a generic type to those implementing a trait.
                      | Syntax: T: SomeTrait ensures, only types with SomeTrait are allowed.
                      | Trait bounds enables access to the trait’s methods inside the function.

Bound Syntax Variants | Trait bounds can be expressed in multiple equivalent forms.
                      | T: Trait, impl Trait, and where clause are alternatives.
                      | where clause improves readability for multiple complex bounds.

impl Trait in Returns | impl Trait can be used as a function return type.
                      | It indicates the function returns some type implementing the trait.
                      | impl Trait return requires one concrete type.
                      | Conditional returns of different concrete types are not allowed.
                      | Trait objects are required for such dynamic return scenarios.
----------------------------------------------------------------------------------------------
*/
