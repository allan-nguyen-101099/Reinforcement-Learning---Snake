// -----------------------------------------------------
// 		Choosing Associated vs Generic Types
// -----------------------------------------------------

// Using GENERICS in the trait (not associated types)
// This allows a type to implement Addition MULTIPLE TIMES with different Rhs and Output types
trait Addition<Rhs, Output> {  // Rhs = type of the right-hand side operand, Output = result type
    fn add(self, rhs: Rhs) -> Output;  // `self` taken by value (moved) — consumes the caller
}

struct Point {
    x: i32,
    y: i32,
}

// Implementation 1: Point + Point = Point
impl Addition<Point, Point> for Point {
    fn add(self, rhs: Point) -> Point {  // Rhs=Point, Output=Point
        Point {
            x: self.x + rhs.x,  // add x coordinates
            y: self.y + rhs.y,  // add y coordinates
        }
    }
}

// Implementation 2: Point + i32 = Point (shift both coords by a scalar)
impl Addition<i32, Point> for Point {  // Rhs=i32 (different!), Output=Point — second impl for same type!
    fn add(self, rhs: i32) -> Point {
        Point {
            x: self.x + rhs,  // add scalar to x
            y: self.y + rhs,  // add scalar to y
        }
    }
}

struct Line {
    start: Point,
    end: Point,
}

// Implementation 3: Point + Point = Line (DIFFERENT output type)
impl Addition<Point, Line> for Point {  // Rhs=Point, Output=Line — third impl for same type!
    fn add(self, rhs: Point) -> Line {   // same Rhs but different Output vs impl 1 — allowed with generics
        Line {
            start: self,  // self becomes the start of the line
            end: rhs,     // rhs becomes the end of the line
        }
    }
}
fn main() {
    let p1 = Point { x: 1, y: 1 };
    let p2 = Point { x: 2, y: 2 };
    let p3: Point = p1.add(p2);

    assert_eq!(p3.x, 3);
    assert_eq!(p3.y, 3);

    let p1 = Point { x: 1, y: 1 };
    let p3 = p1.add(2);

    assert_eq!(p3.x, 3);
    assert_eq!(p3.y, 3);

    let p1 = Point { x: 1, y: 1 };
    let p2 = Point { x: 2, y: 2 };
    let line: Line = p1.add(p2);

    assert!(line.start.x == 1 && line.start.y == 1 && line.end.x == 2 && line.end.y == 2);
}

/*
------------------------------------------------------------------------------------------------------------------
Concept/Topic                | Explanation
-----------------------------|------------------------------------------------------------------------------------
Associated Types Limitation  | Associated types allow only one implementation of a trait per concrete type.
                             | Once associated types are fixed, they cannot vary across implementations.

Using Generics Flexibility   | Converting associated types into generics enables multiple implementations.
                             | Each implementation can choose different generic parameters.

Associated Types vs Generics | Use associated types when a trait has exactly one logical implementation per type.
                             | Use generics when multiple implementations per type are required.
------------------------------------------------------------------------------------------------------------------
 */
