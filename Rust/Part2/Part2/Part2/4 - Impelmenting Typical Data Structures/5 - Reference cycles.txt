use std::borrow::Borrow; 
use std::rc::{Rc, Weak};  // Weak = a non-owning reference — does NOT increase strong_count
                          // Prevents reference cycles: Rc keeps data alive, Weak does not
use std::cell::{RefCell, Ref}; 

#[derive(Debug)]
struct Node {
    value: i32, 
    parent: RefCell<Weak<Node>>,      // Weak pointer to parent — parent does NOT keep children alive
                                       // Using Weak breaks the cycle: child -> parent is Weak, parent -> child is Rc
    children: RefCell<Vec<Rc<Node>>>, // Rc pointers to children — children are OWNED by parent
                                       // RefCell allows mutating children after construction
}
fn main() { 
    // leaf has no parent yet, no children
    let leaf  = Rc::new(Node {
        value: 3, 
        parent: RefCell::new(Weak::new()),  // Weak::new() = empty/unset weak reference
        children: RefCell::new(vec![]),
    }); 

    // branch owns leaf as a child
    let branch = Rc::new(Node {
        value: 5, 
        parent: RefCell::new(Weak::new()),
        children: RefCell::new(vec![Rc::clone(&leaf)]),  // branch strongly owns leaf
    });

    // Set leaf's parent to branch using a WEAK reference
    // Rc::downgrade(&branch) creates a Weak<Node> from an Rc<Node> — no strong_count increment
    *leaf.parent.borrow_mut() = Rc::downgrade(&branch);
    // Now: branch -> leaf is strong (Rc), leaf -> branch is weak (Weak)
    // No cycle: when branch drops, leaf's weak parent becomes invalid — no memory leak
}
